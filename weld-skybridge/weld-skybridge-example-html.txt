<script>
(function () {
  var D = window.__sbDemo;
  function $(id){ return document.getElementById(id); }
  function show(el, v){ el.textContent = (typeof v === 'string') ? v : JSON.stringify(v, null, 2); }

  function paint(s) {
    var c = $('conn'), t = $('conn-txt');
    if (s.connected) { c.className = 'pill on'; t.textContent = 'companion linked'; }
    else { c.className = 'pill off'; t.textContent = 'no companion (fallback)'; }
    $('proto').textContent = 'protocol ' + (s.protocol || '—');
    $('proto').className = 'pill ' + (s.connected ? 'on' : 'off');
    $('backend').textContent = 'storage: ' + s.backend;
    $('backend').className = 'pill ' + (s.backend === 'companion' ? 'on' : 'off');
  }

  function refresh(){ if (D) paint(D.status()); }

  // react to a late connection
  window.__sbOnConnect = function(){ refresh(); };

  if (!D) { $('conn-txt').textContent = 'plugin not loaded'; return; }
  refresh();
  // the handshake is async; check again shortly after load
  setTimeout(refresh, 600);

  $('save').onclick = function(){ D.save($('slot').value, $('text').value).then(function(ok){ show($('store-out'), ok ? ('saved \u2713  (' + D.status().backend + ')') : 'save failed'); refresh(); }); };
  $('load').onclick = function(){ D.load($('slot').value).then(function(v){ show($('store-out'), v ? v : '(nothing in that slot)'); }); };
  $('list').onclick = function(){ D.listSlots().then(function(keys){ show($('store-out'), keys && keys.length ? keys : '(no slots yet)'); }); };

  $('ai-go').onclick = function(){
    show($('ai-out'), 'thinking\u2026');
    D.ai($('ai-in').value).then(function(r){
      if (r.ok) show($('ai-out'), r.value);
      else if (r.reason === 'no-own-model') show($('ai-out'), 'No personal model configured. Open the companion \u2192 AI Helper tab and add your provider + key.');
      else if (r.reason === 'denied') show($('ai-out'), 'Permission denied for this generator.');
      else if (r.reason === 'disconnected' || r.reason === 'unsupported') show($('ai-out'), 'AI needs the Weld Companion userscript (with your own model set up).');
      else show($('ai-out'), 'unavailable: ' + r.reason);
    });
  };
})();
</script>
